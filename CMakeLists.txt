cmake_minimum_required(VERSION 3.21)   # clap-wrapper needs 3.21

# C is needed for DPF's pugl / dgl sources.
# macOS: 26 or later, Apple Silicon only. Set before project();
# ignored on other platforms. Both are cache variables, so -D overrides them.
set(CMAKE_OSX_DEPLOYMENT_TARGET 26.0 CACHE STRING "macOS 26 or later")
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Apple Silicon only")
project(ModicumFM VERSION 0.9.0 LANGUAGES C CXX)

# Everything is linked into the .clap, a shared module, so every static library in it (the engine,
# the plugin core, DPF's DGL) must be position-independent. clang on Linux defaults to PIE and
# hides this; gcc does not ("relocation R_X86_64_PC32 ... recompile with -fPIC"). As IFSynth
# (e5a32e3) and PlainOldSynth.
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

set(CMAKE_CXX_STANDARD 20)          # PM framework uses std::numbers / std::span
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(MODICUMFM_WERROR "Treat warnings as errors (framework + tests)" OFF)
option(MODICUMFM_PRESET_DUMP "Dev-only: press 'P' in the UI to dump the patch as JSON to \$HOME" OFF)
option(MODICUMFM_BUILD_VST3 "Build a VST3 wrapper around the same plugin via clap-wrapper." ON)
option(MODICUMFM_BUILD_AUV2 "Build an AUv2 (.component) wrapper via clap-wrapper (macOS only)." ${APPLE})

if(APPLE)
    # The .mm UI sources are Objective-C++, and clap-wrapper's AUv2 wrapper needs the language enabled.
    enable_language(OBJCXX)
endif()

# --- presets: Presets/*.json compiled in as a C++ array, not loaded at runtime -------------
# Factory presets need no file on disk and don't depend on CLAP's own preset-discovery extension —
# the UI presents its own dropdown. (The UI's Save / Load reads and writes the same JSON for the
# user's own patches: src/UI/PresetFile.hpp.) Adding/removing a preset file requires
# reconfiguring (CONFIGURE_DEPENDS re-globs on build, but a brand new file still needs `cmake -B
# build -S .` rerun once to pick up the new match).
find_package(Python3 COMPONENTS Interpreter REQUIRED)
file(GLOB MODICUMFM_PRESET_FILES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/Presets/*.json")
set(MODICUMFM_GENERATED_PRESETS_H "${CMAKE_BINARY_DIR}/generated/GeneratedPresets.h")
add_custom_command(
    OUTPUT "${MODICUMFM_GENERATED_PRESETS_H}"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tools/gen_presets.py"
            ${MODICUMFM_PRESET_FILES} -o "${MODICUMFM_GENERATED_PRESETS_H}"
    DEPENDS ${MODICUMFM_PRESET_FILES} "${CMAKE_SOURCE_DIR}/tools/gen_presets.py"
    COMMENT "Generating embedded presets from Presets/*.json")
add_custom_target(generate_presets DEPENDS "${MODICUMFM_GENERATED_PRESETS_H}")

# --- embedded resources: Resources/* compiled in as raw bytes ----------------------------
# Uses DPF's own res2c.py (ships in the dpf/ submodule) to embed the logo PNG's raw file bytes;
# NanoVG's createImageFromMemory() decodes it at UI-construction time. No filesystem dependency
# on the resource file once built — same reasoning as the preset codegen above.
set(MODICUMFM_RESOURCES_HPP "${CMAKE_BINARY_DIR}/generated/EmbeddedResources.hpp")
set(MODICUMFM_RESOURCES_CPP "${CMAKE_BINARY_DIR}/generated/EmbeddedResources.cpp")
# res2c.py embeds a whole folder and names each symbol by the file's stem, so help.md (the source of
# help.html) would collide with help.html. Copy everything but *.md into a staging folder first.
file(GLOB MODICUMFM_RESOURCE_FILES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/Resources/*")
list(FILTER MODICUMFM_RESOURCE_FILES EXCLUDE REGEX "\\.md$")
set(MODICUMFM_RESOURCES_STAGING "${CMAKE_BINARY_DIR}/generated/Resources")
add_custom_command(
    OUTPUT "${MODICUMFM_RESOURCES_HPP}" "${MODICUMFM_RESOURCES_CPP}"
    COMMAND ${CMAKE_COMMAND} -E rm -rf "${MODICUMFM_RESOURCES_STAGING}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${MODICUMFM_RESOURCES_STAGING}"
    COMMAND ${CMAKE_COMMAND} -E copy ${MODICUMFM_RESOURCE_FILES} "${MODICUMFM_RESOURCES_STAGING}"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/dpf/utils/res2c.py"
            EmbeddedResources "${MODICUMFM_RESOURCES_STAGING}" "${CMAKE_BINARY_DIR}/generated"
    DEPENDS ${MODICUMFM_RESOURCE_FILES} "${CMAKE_SOURCE_DIR}/dpf/utils/res2c.py"
    COMMENT "Embedding Resources/* as compiled-in data")
add_custom_target(generate_resources DEPENDS "${MODICUMFM_RESOURCES_HPP}" "${MODICUMFM_RESOURCES_CPP}")

# --- DPF is used ONLY for the GUI ------------------------------------------
# UIExporter + NanoVG + pugl. The CLAP entry point
# is ours (src/plugin.cpp), because DPF's generated CLAP wrapper cannot receive MIDI 2.0 note
# events: see that file's header.
add_subdirectory(dpf)
# dgl-opengl is normally created by dpf_add_plugin()'s UI_TYPE branch; call the same helper
# directly. SHARED_RESOURCES=TRUE embeds DGL's DejaVu font. USE_FILE_BROWSER=TRUE builds the file
# dialog Save / Load open (DISTRHO_UI_FILE_BROWSER). USE_WEB_VIEW=FALSE: Help opens in the default
# browser, so no WebView (and no WebKit) is built.
dpf__add_dgl_opengl(TRUE TRUE FALSE)
# ModicumFM's own namespaces: on macOS pugl names its Objective-C classes after DGL_NAMESPACE, and
# DPF's file dialog its delegate after DISTRHO_NAMESPACE, so two DPF binaries
# that share either in one host collide. PUBLIC, so every DPF source in the plugin agrees.
target_compile_definitions(dgl-opengl PUBLIC DGL_NAMESPACE=ModicumFMDGL DISTRHO_NAMESPACE=ModicumFMDISTRHO)

# --- modicumfm_core: the whole plugin (DSP + host glue + GUI shell) as one static archive --------
# Hidden visibility for our own objects, so clap_entry (marked CLAP_EXPORT in plugin.cpp) is the
# only symbol this target exports (the vendored engine and DGL libraries keep their own default
# visibility).
add_library(modicumfm_core STATIC
    src/plugin.cpp
    src/UI/ModicumFMUI.cpp
    "${MODICUMFM_RESOURCES_CPP}"                # generated: the logo/help bytes (see above)
    "${DPF_ROOT_DIR}/distrho/src/DistrhoUI.cpp"
    "${DPF_ROOT_DIR}/distrho/src/DistrhoUtils.cpp")
set_target_properties(modicumfm_core PROPERTIES
    C_VISIBILITY_PRESET hidden
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON)
target_include_directories(modicumfm_core PUBLIC
    src                               # plugin_gui.h (the DSP <-> GUI seam)
    plugin                            # DistrhoPluginInfo.h + ModicumFMParams.h
    external/clap/include             # <clap/clap.h> (the raw entry point)
    "${CMAKE_BINARY_DIR}/generated"   # GeneratedPresets.h + EmbeddedResources.hpp
    "${DPF_ROOT_DIR}/distrho")        # DistrhoUI.hpp / DistrhoUIInternal.hpp
# clap-wrapper's plugin-side extension header only (clapwrapper/vst3.h: which VST3 note expressions to
# offer). Header-only and plain C, so the CLAP build needs it even with MODICUMFM_BUILD_VST3 off.
target_include_directories(modicumfm_core PRIVATE "${CMAKE_SOURCE_DIR}/external/clap-wrapper/include")
target_compile_definitions(modicumfm_core PUBLIC
    DISTRHO_PLUGIN_TARGET_CLAP
    DISTRHO_IS_STANDALONE=0
    MODICUMFM_VERSION_STRING="${PROJECT_VERSION}")
target_link_libraries(modicumfm_core PUBLIC pm_framework dgl-opengl)
add_dependencies(modicumfm_core generate_presets generate_resources)
# FALSE = no WebView. On macOS this still adds DistrhoUI_macOS.mm (the editor's native window glue).
dpf__add_plugin_specific_ui_sources(modicumfm_core FALSE)
if(APPLE)
    # The live Retina scale of the host window (see the file's header comment).
    target_sources(modicumfm_core PRIVATE src/UI/BackingScale_mac.mm)
endif()
# Dev-only JSON patch dumper (keystroke → file); compiled out of release builds.
if(MODICUMFM_PRESET_DUMP)
    target_compile_definitions(modicumfm_core PRIVATE MODICUMFM_PRESET_DUMP=1)
endif()

# --- the .clap output ------------------------------------------------------
# A MODULE with no sources of its own would be rejected by CMake, and simply linking the archive is
# not enough: nothing references clap_entry, so the linker would drop the object that defines it.
# Hence the empty placeholder plus a whole-archive force-load.
function(modicumfm_force_load_whole_archive TARGET_NAME LIB_NAME)
    if(APPLE)
        target_link_options(${TARGET_NAME} PRIVATE "-Wl,-force_load,$<TARGET_FILE:${LIB_NAME}>")
    elseif(UNIX)
        target_link_options(${TARGET_NAME} PRIVATE
            "-Wl,--whole-archive" "$<TARGET_FILE:${LIB_NAME}>" "-Wl,--no-whole-archive")
    endif()
endfunction()

set(MODICUMFM_EMPTY_SOURCE "${CMAKE_BINARY_DIR}/generated/modicumfm_empty.c")
# A genuinely empty file trips -Wempty-translation-unit under -Wpedantic -Werror, and ISO C requires
# at least one declaration anyway.
set(_modicumfm_empty_content "typedef int modicumfm_empty_translation_unit_marker;\n")
set(_modicumfm_empty_existing "")
if(EXISTS "${MODICUMFM_EMPTY_SOURCE}")
    file(READ "${MODICUMFM_EMPTY_SOURCE}" _modicumfm_empty_existing)
endif()
if(NOT _modicumfm_empty_existing STREQUAL _modicumfm_empty_content)
    file(WRITE "${MODICUMFM_EMPTY_SOURCE}" "${_modicumfm_empty_content}")
endif()

add_library(ModicumFM MODULE "${MODICUMFM_EMPTY_SOURCE}")
target_link_libraries(ModicumFM PRIVATE modicumfm_core)
modicumfm_force_load_whole_archive(ModicumFM modicumfm_core)

if(APPLE)
    set_target_properties(ModicumFM PROPERTIES
        BUNDLE TRUE
        BUNDLE_EXTENSION "clap"
        MACOSX_BUNDLE_GUI_IDENTIFIER "com.unlikelyware.modicumfm"
        MACOSX_BUNDLE_BUNDLE_NAME "ModicumFM"
        MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
        MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
else()
    set_target_properties(ModicumFM PROPERTIES PREFIX "" SUFFIX ".clap")
endif()

# --- install-clap: symlink the built .clap into the user CLAP dir ----------
# Only when asked (`cmake --build build --target install-clap`): an ordinary build never writes
# outside build/ (ReleaseUMP R1-36). The link points at the freshest build after a rescan. The rm
# first clears an old link or a copied bundle directory, which create_symlink would not replace.
if(APPLE)
    set(MODICUMFM_CLAP_DIR "$ENV{HOME}/Library/Audio/Plug-Ins/CLAP")
    set(MODICUMFM_CLAP_ARTIFACT "$<TARGET_BUNDLE_DIR:ModicumFM>")
else()
    set(MODICUMFM_CLAP_DIR "$ENV{HOME}/.clap")
    set(MODICUMFM_CLAP_ARTIFACT "$<TARGET_FILE:ModicumFM>")
endif()
add_custom_target(install-clap
    COMMAND ${CMAKE_COMMAND} -E make_directory "${MODICUMFM_CLAP_DIR}"
    COMMAND ${CMAKE_COMMAND} -E rm -rf "${MODICUMFM_CLAP_DIR}/ModicumFM.clap"
    COMMAND ${CMAKE_COMMAND} -E create_symlink
        "${MODICUMFM_CLAP_ARTIFACT}"
        "${MODICUMFM_CLAP_DIR}/ModicumFM.clap"
    DEPENDS ModicumFM
    COMMENT "Symlinking ModicumFM.clap into ${MODICUMFM_CLAP_DIR}"
    VERBATIM)

# --- VST3 / AUv2 via the vendored clap-wrapper, wrapping the very same CLAP -------------------
# Purely additive: nothing here touches the .clap target above. Like the .clap, these are
# installed only by their targets, `install-vst3` / `install-auv2`.
if(MODICUMFM_BUILD_VST3 OR MODICUMFM_BUILD_AUV2)
    if(NOT CLAP_SDK_ROOT)
        set(CLAP_SDK_ROOT "${CMAKE_SOURCE_DIR}/external/clap" CACHE PATH "" FORCE)
    endif()
    add_subdirectory(external/clap-wrapper)
endif()

if(MODICUMFM_BUILD_VST3)
    if(NOT VST3_SDK_ROOT)
        set(VST3_SDK_ROOT "${CMAKE_SOURCE_DIR}/external/vst3sdk" CACHE PATH "" FORCE)
    endif()

    add_library(ModicumFM_vst3 MODULE "${MODICUMFM_EMPTY_SOURCE}")
    if(UNIX AND NOT APPLE)
        # clap-wrapper's Linux branch tests a misspelled TCLP_RESOURCE_DIRECTORY, which is never
        # set, so it always warns that RESOURCE_DIRECTORY is unsupported. Defining it empty
        # silences that; we pass no RESOURCE_DIRECTORY.
        set(TCLP_RESOURCE_DIRECTORY "")
    endif()
    target_link_libraries(ModicumFM_vst3 PRIVATE modicumfm_core)
    modicumfm_force_load_whole_archive(ModicumFM_vst3 modicumfm_core)
    target_add_vst3_wrapper(
        TARGET ModicumFM_vst3
        OUTPUT_NAME "ModicumFM"
        BUNDLE_IDENTIFIER "com.unlikelyware.modicumfm.vst3"
        BUNDLE_VERSION "${PROJECT_VERSION}")
    if(TARGET base-sdk-vst3 AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        # The VST3 SDK still uses std::wstring_convert, deprecated since C++17.
        target_compile_options(base-sdk-vst3 PRIVATE -Wno-deprecated-declarations)
    endif()

    if(APPLE)
        set(MODICUMFM_VST3_DIR "$ENV{HOME}/Library/Audio/Plug-Ins/VST3")
        set(MODICUMFM_VST3_ARTIFACT "$<TARGET_BUNDLE_DIR:ModicumFM_vst3>")
    else()
        # On Linux $<TARGET_FILE> is the .so inside the bundle; hosts want the bundle folder,
        # which clap-wrapper builds under build/Release (or build/Debug).
        set(MODICUMFM_VST3_DIR "$ENV{HOME}/.vst3")
        set(MODICUMFM_VST3_ARTIFACT
            "${CMAKE_BINARY_DIR}/$<IF:$<CONFIG:Debug>,Debug,Release>/ModicumFM.vst3")
    endif()
    add_custom_target(install-vst3
        COMMAND ${CMAKE_COMMAND} -E make_directory "${MODICUMFM_VST3_DIR}"
        COMMAND ${CMAKE_COMMAND} -E rm -rf "${MODICUMFM_VST3_DIR}/ModicumFM.vst3"
        COMMAND ${CMAKE_COMMAND} -E create_symlink "${MODICUMFM_VST3_ARTIFACT}" "${MODICUMFM_VST3_DIR}/ModicumFM.vst3"
        DEPENDS ModicumFM_vst3
        VERBATIM
        COMMENT "Symlinking ModicumFM.vst3 into ${MODICUMFM_VST3_DIR}")
endif()

if(MODICUMFM_BUILD_AUV2)
    # AUv2 is an Apple-only format, and clap-wrapper's target_add_auv2_wrapper does not exist off
    # Apple platforms. Say so plainly instead of failing with an unknown-command error.
    if(NOT APPLE)
        message(FATAL_ERROR
            "MODICUMFM_BUILD_AUV2 is macOS-only (AUv2 is an Apple plugin format). "
            "Reconfigure with -DMODICUMFM_BUILD_AUV2=OFF.")
    endif()
    # clap-wrapper fetches Apple's AudioUnitSDK itself (CPM) on the first configure, unless pointed
    # at a local copy with -DAUDIOUNIT_SDK_ROOT=...
    if(NOT AUDIOUNIT_SDK_ROOT)
        set(CLAP_WRAPPER_DOWNLOAD_DEPENDENCIES ON)
    endif()

    add_library(ModicumFM_auv2 MODULE "${MODICUMFM_EMPTY_SOURCE}")
    target_link_libraries(ModicumFM_auv2 PRIVATE modicumfm_core)
    modicumfm_force_load_whole_archive(ModicumFM_auv2 modicumfm_core)
    # "MdFM" (ModicumFM) as the subtype, under Unlikelyware's own "nLKL" manufacturer code.
    target_add_auv2_wrapper(
        TARGET ModicumFM_auv2
        OUTPUT_NAME "ModicumFM"
        BUNDLE_IDENTIFIER "com.unlikelyware.modicumfm.auv2"
        BUNDLE_VERSION "${PROJECT_VERSION}"
        MANUFACTURER_NAME "Unlikelyware"
        MANUFACTURER_CODE "nLKL"
        SUBTYPE_CODE "MdFM"
        INSTRUMENT_TYPE "aumu")

    # A copy, not a symlink: AU hosts don't reliably find symlinked components.
    set(MODICUMFM_AUV2_DIR "$ENV{HOME}/Library/Audio/Plug-Ins/Components")
    add_custom_target(install-auv2
        COMMAND ${CMAKE_COMMAND} -E make_directory "${MODICUMFM_AUV2_DIR}"
        COMMAND ${CMAKE_COMMAND} -E rm -rf "${MODICUMFM_AUV2_DIR}/ModicumFM.component"
        COMMAND ${CMAKE_COMMAND} -E copy_directory
                "$<TARGET_BUNDLE_DIR:ModicumFM_auv2>" "${MODICUMFM_AUV2_DIR}/ModicumFM.component"
        DEPENDS ModicumFM_auv2
        VERBATIM
        COMMENT "Copying ModicumFM.component into ${MODICUMFM_AUV2_DIR}")
endif()
